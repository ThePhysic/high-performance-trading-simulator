cmake_minimum_required(VERSION 3.20)
project(high_performance_trading_simulator VERSION 0.1.0 LANGUAGES CXX)

set(HPTS_SANITIZERS "" CACHE STRING "Sanitizers: address,undefined or thread (compiler dependent)")
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(hpts_engine src/engine.cpp src/io.cpp src/files.cpp)
target_include_directories(hpts_engine PUBLIC include)
target_compile_features(hpts_engine PUBLIC cxx_std_20)
target_link_libraries(hpts_engine PUBLIC Threads::Threads)

add_executable(hpts src/main.cpp)
target_link_libraries(hpts PRIVATE hpts_engine)
if(WIN32)
    target_link_libraries(hpts PRIVATE Psapi)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(hpts_engine_tests tests/test_engine.cpp)
    target_link_libraries(hpts_engine_tests PRIVATE hpts_engine)
    add_test(NAME engine_contract COMMAND hpts_engine_tests)
    add_executable(hpts_queue_tests tests/test_queue.cpp)
    target_link_libraries(hpts_queue_tests PRIVATE hpts_engine)
    add_test(NAME queue_contract COMMAND hpts_queue_tests)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME cli_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:hpts>" "${CMAKE_CURRENT_SOURCE_DIR}")
    endif()
endif()

set(hpts_targets hpts_engine hpts)
if(BUILD_TESTING)
    list(APPEND hpts_targets hpts_engine_tests hpts_queue_tests)
endif()
foreach(target IN LISTS hpts_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
        if(HPTS_SANITIZERS)
            if(NOT HPTS_SANITIZERS STREQUAL "address")
                message(FATAL_ERROR "MSVC supports only HPTS_SANITIZERS=address")
            endif()
            target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
            target_link_options(${target} PRIVATE /INCREMENTAL:NO)
        endif()
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(HPTS_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=${HPTS_SANITIZERS} -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=${HPTS_SANITIZERS})
        endif()
    endif()
endforeach()
